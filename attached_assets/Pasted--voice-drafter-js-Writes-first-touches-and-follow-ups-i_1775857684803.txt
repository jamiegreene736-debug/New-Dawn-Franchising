// voice-drafter.js
// Writes first touches and follow-ups in Dylan's voice.
// The voice profile is stored in the DB and hot-swappable — edit it
// in the UI, and the next draft picks up the changes without a deploy.

import Anthropic from "@anthropic-ai/sdk";
import { z } from "zod";

const client = new Anthropic();
const MODEL = "claude-sonnet-4-5";

// ---------- Output schemas ----------
const EmailDraftSchema = z.object({
  subject: z.string().min(3).max(80),
  body: z.string().min(40),
  personalization_hook: z.string().min(10),
  confidence_score: z.number().min(0).max(100),
  ymyl_flag: z.boolean(),
  notes_for_human: z.string().optional(),
});

const WhatsAppDraftSchema = z.object({
  template_name: z.string().nullable(), // required for first touch, null for replies
  body: z.string().min(20).max(1024),
  personalization_hook: z.string().min(10),
  confidence_score: z.number().min(0).max(100),
  notes_for_human: z.string().optional(),
});

const LinkedInDraftSchema = z.object({
  body: z.string().min(40).max(300), // LinkedIn DM character sweet spot
  personalization_hook: z.string().min(10),
  confidence_score: z.number().min(0).max(100),
  notes_for_human: z.string().optional(),
});

// ---------- System prompt builder ----------
function buildVoiceSystemPrompt(voiceProfile, channel) {
  const samples = voiceProfile.sample_messages
    ?.map((s, i) => `SAMPLE ${i + 1}:\n${s}`)
    .join("\n\n") ?? "(no samples yet — use warm, direct, human tone)";

  const doNotSay = voiceProfile.do_not_say?.length
    ? voiceProfile.do_not_say.join(", ")
    : "";

  const channelRules = {
    email: `
CHANNEL: EMAIL
- Under 120 words in the body
- Subject line under 55 characters, no clickbait, no ALL CAPS, no emoji
- One clear ask, and that ask is a reply — not a meeting (meetings come in touch 2 or 3)
- Must end with the signature block verbatim:
---
${voiceProfile.signature_block ?? "(signature not configured — flag ymyl_flag=true)"}
---`,
    whatsapp: `
CHANNEL: WHATSAPP
- Under 400 characters
- No links in the first touch (Meta flags them)
- First touches MUST fit an approved Business template — if you're not sure which template applies, set template_name to null and flag notes_for_human
- Conversational, not formal — WhatsApp is a personal channel
- No signature block`,
    linkedin_draft: `
CHANNEL: LINKEDIN DM (draft only — a human will send it)
- Under 280 characters (the sweet spot for LinkedIn reply rates)
- No links (LinkedIn deprioritizes messages with links)
- Reference something from their profile or recent activity
- End with a low-friction question
- No signature block`,
  };

  return `You are writing as Dylan, founder of New Dawn Franchising. Your job is to draft outreach messages that sound indistinguishable from messages Dylan writes himself.

STUDY THESE REAL SAMPLES OF DYLAN'S WRITING — match the cadence, sentence length, vocabulary, warmth, and rhythm:

${samples}

TONE RULES (from Dylan):
${voiceProfile.tone_rules ?? "Warm, direct, specific, occasional dry humor, never corporate."}

BANNED PHRASES (never use any of these, or close variants):
- "I hope this email finds you well"
- "reaching out" / "reaching out to"
- "circle back" / "touch base" / "quick question"
- "synergies" / "leverage" / "unlock" / "navigate the landscape"
- "per my last email" / "just wanted to" / "following up on my previous"
- "delve" / "dive deep" / "in today's fast-paced world"
- "It's not just X, it's Y" construction
- Em-dash overuse (one per message maximum)
${doNotSay ? `- ${doNotSay}` : ""}

HARD REQUIREMENTS FOR EVERY FIRST TOUCH:
- Open with something specific about THIS recipient (a post, a case, a firm detail, a mutual connection) — generic openers are an automatic rejection
- State your reason for writing in one sentence
- Offer concrete value BEFORE asking for anything
- Low-friction ask: a reply, an opinion, a quick yes/no — not a meeting
- Truthful at all times. You are writing in Dylan's voice, but you will never invent facts, claim things about New Dawn that aren't established, or mislead about prior contact
${channelRules[channel]}

YMYL FLAG: Set ymyl_flag=true (emails only) if the message touches immigration law, visa eligibility, tax, or legal outcomes in a way that could be interpreted as advice. A flagged draft requires human legal review before sending.

CONFIDENCE SCORE: 0–100, honest assessment of how likely this specific message is to get a reply. Below 60 means the personalization is weak or the angle is off — flag it and explain in notes_for_human so Dylan can improve the lead data or kill the draft.

Return ONLY valid JSON matching the requested schema. No markdown fences, no prose, no commentary outside the JSON.`;
}

// ---------- Lead context builder ----------
function buildLeadContext(lead, campaign, sequenceHistory) {
  const history = sequenceHistory?.length
    ? sequenceHistory
        .map(
          (t, i) =>
            `  Touch ${i + 1} (${t.channel}, ${t.status}): ${t.body?.slice(0, 200) ?? "(no body)"}${t.reply_body ? `\n  → THEIR REPLY: ${t.reply_body.slice(0, 300)}` : ""}`
        )
        .join("\n\n")
    : "  (no prior touches — this is a first touch)";

  return `LEAD
  Name: ${lead.first_name} ${lead.last_name}
  Title: ${lead.job_title}
  Firm: ${lead.firm_name}
  Location: ${lead.city ?? ""}${lead.city && lead.country ? ", " : ""}${lead.country ?? ""}
  LinkedIn: ${lead.linkedin_url ?? "(not available)"}
  Website: ${lead.website_url ?? "(not available)"}
  Lead score: ${lead.lead_score ?? "unscored"}
  Score rationale: ${lead.score_rationale ?? "(none)"}

PERSONALIZATION INTEL (use something specific from here — if nothing is specific enough, set confidence_score below 60 and explain in notes_for_human):
${lead.personalization_intel ?? "  (none gathered — weak personalization ahead)"}

CAMPAIGN CONTEXT
  Why Dylan wants to talk to this persona: ${campaign.persona_reason ?? "Referral partner for E-2 visa franchise clients"}
  What Dylan offers that's valuable to them: ${campaign.value_proposition ?? "(not configured)"}

SEQUENCE HISTORY
${history}`;
}

// ---------- Main drafter ----------
export async function draftMessage({
  lead,
  campaign,
  voiceProfile,
  channel, // "email" | "whatsapp" | "linkedin_draft"
  touchType, // "first_touch" | "follow_up" | "reply_response"
  sequenceHistory = [],
  replyToHandle = null, // if touchType === "reply_response"
  angle = null, // optional guidance from the planner
}) {
  const systemPrompt = buildVoiceSystemPrompt(voiceProfile, channel);
  const leadContext = buildLeadContext(lead, campaign, sequenceHistory);

  const instruction =
    touchType === "first_touch"
      ? `Write a first touch for this lead. Angle suggested by the planner: ${angle ?? "(you choose the strongest angle based on the intel above)"}.`
      : touchType === "follow_up"
        ? `Write a follow-up. Reference the prior touch naturally without being pushy. Do not say "just following up" or "per my last email" or anything like it. Offer new value or a new angle.`
        : `Write a response to this reply from the lead:\n\nTHEIR REPLY: ${replyToHandle}\n\nMatch their energy and length. If they're interested, propose the Calendly link naturally. If they have an objection, address it directly and honestly. If they said no, thank them warmly and close the sequence.`;

  const userMessage = `${leadContext}\n\n${instruction}\n\nDraft the message now and return the JSON.`;

  let lastError;
  for (let attempt = 1; attempt <= 3; attempt++) {
    try {
      const response = await client.messages.create({
        model: MODEL,
        max_tokens: 1024,
        system: systemPrompt,
        messages: [{ role: "user", content: userMessage }],
      });

      const text = response.content
        .filter((b) => b.type === "text")
        .map((b) => b.text)
        .join("")
        .trim();

      const cleaned = text.replace(/^```(?:json)?\s*/i, "").replace(/\s*```$/i, "");
      const parsed = JSON.parse(cleaned);

      const schema =
        channel === "email"
          ? EmailDraftSchema
          : channel === "whatsapp"
            ? WhatsAppDraftSchema
            : LinkedInDraftSchema;

      const validated = schema.parse(parsed);

      // Post-validation lint: catch banned phrases the model slipped through
      const lintIssues = lintForBannedPhrases(validated.body, voiceProfile);
      if (lintIssues.length > 0 && attempt < 3) {
        // Retry with explicit feedback
        lastError = new Error(`Lint failed: ${lintIssues.join(", ")}`);
        continue;
      }

      return {
        success: true,
        draft: validated,
        channel,
        lint_warnings: lintIssues,
        usage: response.usage,
      };
    } catch (err) {
      lastError = err;
      console.warn(`Drafter attempt ${attempt} failed:`, err.message);
    }
  }

  return {
    success: false,
    draft: null,
    error: lastError?.message ?? "drafter failed",
  };
}

// ---------- Banned-phrase lint ----------
const GLOBAL_BANNED = [
  /\bhope this (email|message) finds you well\b/i,
  /\bjust (wanted|wanted to|reaching out|checking in)\b/i,
  /\bcircle back\b/i,
  /\btouch base\b/i,
  /\bper my last (email|message)\b/i,
  /\bsynerg(y|ies)\b/i,
  /\bleverage\b/i,
  /\bdelve\b/i,
  /\bunlock the\b/i,
  /\bnavigat(e|ing) the landscape\b/i,
  /\bin today'?s fast-paced\b/i,
  /it'?s not just .{1,40}, it'?s\b/i,
];

function lintForBannedPhrases(body, voiceProfile) {
  const issues = [];
  for (const pattern of GLOBAL_BANNED) {
    if (pattern.test(body)) issues.push(`banned phrase matched: ${pattern}`);
  }
  // Check per-profile custom bans
  for (const phrase of voiceProfile.do_not_say ?? []) {
    if (body.toLowerCase().includes(phrase.toLowerCase())) {
      issues.push(`custom banned phrase: "${phrase}"`);
    }
  }
  // Em-dash overuse (more than one)
  const emDashCount = (body.match(/—/g) ?? []).length;
  if (emDashCount > 1) issues.push(`em-dash overuse: ${emDashCount} found`);
  return issues;
}
