3. Add Apollo's organization search to apollo.js
javascriptexport async function apolloSearchByCompany(companyName, titles) {
  const res = await fetch("https://api.apollo.io/v1/mixed_people/search", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      api_key: APOLLO_KEY,
      q_organization_name: companyName,
      person_titles: titles,
      page: 1,
      per_page: 25,
    }),
  });
  const data = await res.json();
  return (data.people || []).map(p => ({
    first_name: p.first_name,
    last_name: p.last_name,
    full_name: p.name,
    title: p.title,
    company: companyName,
    email: p.email,
    phone: p.phone_numbers?.[0]?.sanitized_number,
    linkedin_url: p.linkedin_url,
    source: "apollo_company_search",
  }));
}

5. Try Hunter.io's domain search (not email finder)
Hunter has a separate endpoint that returns all emails it has on a domain — including the owner's. Add this:
javascript// services/hunter.js — add this function
export async function hunterDomainSearch(domain) {
  const res = await fetch(
    `https://api.hunter.io/v2/domain-search?domain=${domain}&api_key=${HUNTER_KEY}&limit=100`
  );
  const data = await res.json();
  return data?.data?.emails || [];
}
This often returns the founder's email even when name-based searches miss them.
